// Two players' RemoteHosts, the driver's own src/driver/remote-host.ts, play one round against `wrangler dev`, then fill the league and the online cap.
// Usage: node e2e.mjs <worktree> <backendUrl>
import { randomBytes } from "node:crypto";
import { pathToFileURL } from "node:url";

const [worktree, backendUrl] = process.argv.slice(2);
const { createRemoteHost } = await import(pathToFileURL(`${worktree}/src/driver/remote-host.ts`).href);
const { deriveRoomId, PROTOCOL_VERSION } = await import(pathToFileURL(`${worktree}/src/shared/protocol.ts`).href);

console.log = () => {};
const started = Date.now();
const out = (line) => process.stdout.write(`[+${String(Date.now() - started).padStart(4)} ms] ${line}\n`);
const check = (ok, what) => {
    out(`${ok ? "PASS" : "FAIL"} ${what}`);
    if (!ok) process.exit(1);
};

const joinToken = randomBytes(32).toString("hex");
const roomId = deriveRoomId(joinToken);
out(`backend ${backendUrl}, node ${process.version}, protocol v${PROTOCOL_VERSION}`);
out(`joinToken ${joinToken.slice(0, 12)}…, roomId = deriveRoomId(joinToken) = ${roomId}`);

const describe = (m) => {
    if (m.type === "result") return `result ${m.command} ${m.requestId} -> ${m.code} (r${m.revision})`;
    if (m.type === "fault") return `fault ${m.code}`;
    const c = m.room.current;
    const round = c ? `${c.status} ${c.id.slice(0, 8)}${c.targets ? ` finds ${c.finds.length}/${c.targets.length}` : ""}${c.reason ? ` ${c.reason}` : ""}` : "no round";
    const league = m.room.league.map((e) => `${e.playerName} ${e.finds}`).join(", ");
    return `snapshot r${m.revision}: ${round}, difficulty ${m.room.difficulty}, nextDifficulty ${m.room.nextDifficulty}, league [${league}]`;
};

function client(name, color) {
    const host = createRemoteHost({ backendUrl, roomId });
    const messages = [];
    const waiters = [];
    let seq = 0;
    const self = {
        name,
        messages,
        latest: () => messages.findLast((m) => m.type === "snapshot"),
        open: () =>
            host.open({ v: 1, type: "hello", roomId, joinToken, player: { playerId: `e2e-${name.toLowerCase()}`, name, color } }, (m) => {
                messages.push(m);
                out(`${name} <- ${describe(m)}`);
                for (const w of [...waiters]) if (w.match(m)) waiters.splice(waiters.indexOf(w), 1) && w.resolve(m);
            }),
        until(match, what, ms = 5_000) {
            const seen = messages.find(match);
            if (seen) return Promise.resolve(seen);
            return new Promise((resolve, reject) => {
                const timer = setTimeout(() => reject(new Error(`${name} waited ${ms} ms for ${what}`)), ms);
                waiters.push({ match, resolve: (m) => (clearTimeout(timer), resolve(m)) });
            });
        },
        async send(command) {
            const requestId = `${name.toLowerCase()}-${seq++}`;
            out(`${name} -> ${command.type}${command.targetId ? ` ${command.targetId}` : ""} (${requestId})`);
            host.send({ v: 1, requestId, ...command });
            return self.until((m) => m.type === "result" && m.requestId === requestId, `the result of ${requestId}`);
        },
        close: () => host.close(),
    };
    return self;
}

const b = client("Ben", "#22c55e");
await a.open();
await b.open();
check(a.latest()?.room.roomId === roomId && b.latest()?.room.roomId === roomId, "both hellos admitted with the token-derived room id");

const frameRect = { x: 0, y: 0, width: 800, height: 600 };
const items = [0, 1, 2].map((i) => {
    const x = 80 + i * 230;
    const points = Array.from({ length: 12 }, (_, k) => [x + k * 12, 200 + (k % 2) * 120]);
    return { rect: { x, y: 200, width: 132, height: 120 }, kind: "freedraw", strokeColor: "#e4e4e7", strokeWidth: 2, points };
});

const [liveA, liveB] = await Promise.all([a.until(isLive, "a live snapshot"), b.until(isLive, "a live snapshot")]);
const round = liveA.room.current;

check((await a.send({ type: "find.claim", roundId: round.id, targetId: first.id })).code === "ok", `Ana claims ${first.id}: ok`);

for (const [i, target] of rest.entries()) {
    const player = i % 2 === 0 ? b : a;
    check((await player.send({ type: "find.claim", roundId: round.id, targetId: target.id })).code === "ok", `${player.name} claims ${target.id}: ok`);
}

const isAllFound = (m) => m.type === "snapshot" && m.room.current?.status === "ended" && m.room.current.reason === "all-found";
const [endA, endB] = await Promise.all([a.until(isAllFound, "the all-found snapshot"), b.until(isAllFound, "the all-found snapshot")]);
check(endA.room.difficulty === 1 && endB.room.difficulty === 1, "room.difficulty stays the level just played, 1");
check(endA.room.nextDifficulty === 2 && endB.room.nextDifficulty === 2, "room.nextDifficulty is 2 after all-found");
const late = await b.send({ type: "find.claim", roundId: round.id, targetId: first.id });
check(late.code === "not-live", "a claim after all-found answers not-live");

a.close();
b.close();
out("Ana and Ben closed");

const { MAX_LEAGUE } = await import(pathToFileURL(`${worktree}/src/shared/protocol.ts`).href);
const join = async (who) => {
    let dials = 0;
    const connect = (url) => (dials++, new WebSocket(url));
    const host = createRemoteHost({ backendUrl, roomId, connect });
    let last = null;
    const hello = { v: 1, type: "hello", roomId, joinToken, player: { playerId: `e2e-${who}`, name: who, color: "#698cf9" } };
    const opened = host.open(hello, (m) => (last = m)).then(() => "opened", (error) => error.code);
    const outcome = await opened;
    host.close();
    return { outcome, last, dials: () => dials };
};

    if (outcome !== "opened") check(false, `player ${i} of ${MAX_LEAGUE} joins`);
}
out(`players 3..${MAX_LEAGUE} each opened and closed`);

check(extra.dials() === 1, "it dialed once and did not redial");
const back = await join("ana");
check(back.outcome === "opened", "Ana, already listed, still rejoins the full league");

const MAX_PLAYERS_ONLINE = 32;
const helloFor = (who) => ({ v: 1, type: "hello", roomId, joinToken, player: { playerId: `e2e-${who}`, name: who, color: "#698cf9" } });
const online = [];
for (let i = 3; i < 3 + MAX_PLAYERS_ONLINE; i++) {
    const host = createRemoteHost({ backendUrl, roomId });
    await host.open(helloFor(`p${i}`), () => {});
    online.push(host);
}
out(`players 3..${2 + MAX_PLAYERS_ONLINE} are online at once, ${MAX_PLAYERS_ONLINE} listed players`);
const busyWho = `p${3 + MAX_PLAYERS_ONLINE}`;
let busyDials = 0;
let openedAt = null;
const faults = [];
const busy = createRemoteHost({ backendUrl, roomId, connect: (url) => (busyDials++, new WebSocket(url)) });
const t1 = Date.now();
const busyOpen = busy
    .open(helloFor(busyWho), (m) => {
        if (m.type !== "fault") return;
        faults.push(m);
        out(`${busyWho} <- ${describe(m)}, retryAfterMs ${m.retryAfterMs}`);
    })
    .then(
        () => ((openedAt = Date.now()), "opened"),
        (error) => error.code,
    );
await new Promise((resolve) => setTimeout(resolve, 2_000));
check(faults.length === 1 && faults[0].code === "unavailable", `${busyWho}, listed but one over the online cap, is told unavailable, not full`);
check(busyDials === 1 && openedAt === null, "its RemoteHost keeps its open pending and has not redialed yet");
online.shift().close();
out("p3 goes offline");
const busyOutcome = await busyOpen;
check(busyOutcome === "opened" && busyDials === 2, `it redialed once and got in ${openedAt - t1} ms after its first dial`);
check(openedAt - t1 >= faults[0].retryAfterMs, `no sooner than the room's retryAfterMs, past the 15 s open timeout`);
busy.close();
for (const host of online) host.close();
out("done");
process.exit(0);
